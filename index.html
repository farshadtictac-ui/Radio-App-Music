/* ===================================================================== تنظیمات مخزن گیت‌هاب — باید با تنظیمات داخل index.html یکی باشد ===================================================================== */ const GITHUB_OWNER = "farshadtictac-ui"; const GITHUB_REPO = "Radio-App-Music"; const GITHUB_BRANCH = "main"; const SONGS_PATH = "songs"; // حداکثر حجم فایلی که تلگرام به ربات‌ها اجازه‌ی دانلودش را می‌دهد (محدودیت خودِ تلگرام است) const MAX_BYTES = 19 * 1024 * 1024; function arrayBufferToBase64(buffer) { let binary = ""; const bytes = new Uint8Array(buffer); const chunkSize = 0x8000; // برای جلوگیری از خطای حافظه، فایل تکه‌تکه پردازش می‌شود for (let i = 0; i < bytes.length; i += chunkSize) { binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunkSize)); } return btoa(binary); } // نام فایل امن می‌سازد: یک برچسب زمانی نامرئی جلوی اسم (برای مرتب‌سازی جدید-به-قدیم) // به‌این‌شکل: «۱۷۲۳۴۵۶_خواننده - نام آهنگ [شناسه‌ی‌کوتاه].mp3» function buildFileName(artist, title, uniqueId, timestamp) { const clean = (s) => (s || "").replace(/[\\/:*?"<>|]/g, "").trim(); const base = clean(artist) ? `${clean(artist)} - ${clean(title || "بدون عنوان")}` : clean(title || "بدون عنوان"); const shortId = uniqueId.replace(/[^a-zA-Z0-9]/g, "").slice(-6); const sortKey = String(9999999999 - timestamp); // عدد معکوس تا الفبایی هم جدید-به-قدیم شود return `${sortKey}_${base} [${shortId}].mp3`; } export default { async fetch(request, env, ctx) { const url = new URL(request.url); // ==== بررسی سریع تنظیمات (برای عیب‌یابی) ==== if (request.method === "GET" && url.pathname === "/debug") { const debugLog = JSON.parse((await env.SONGS.get("debugLog")) || "[]"); return new Response(JSON.stringify({ channelIdStored: env.CHANNEL_ID || null, hasBotToken: Boolean(env.BOT_TOKEN), hasWebhookSecret: Boolean(env.WEBHOOK_SECRET), hasGithubToken: Boolean(env.GITHUB_TOKEN), recentEvents: debugLog }, null, 2), { headers: { "Content-Type": "application/json" } }); } // ==== دریافت پیام‌های جدید از کانال (وب‌هوک تلگرام) ==== if (request.method === "POST" && url.pathname === "/webhook") { const secret = request.headers.get("X-Telegram-Bot-Api-Secret-Token"); if (secret !== env.WEBHOOK_SECRET) { return new Response("forbidden", { status: 403 }); } const update = await request.json(); const msg = update.channel_post || update.message; const incomingChatId = msg && msg.chat ? String(msg.chat.id) : null; const matched = incomingChatId !== null && incomingChatId === env.CHANNEL_ID; const hasAudio = Boolean(msg && msg.audio); // ثبت آخرین چند رویداد ورودی برای عیب‌یابی از طریق /debug let debugLog = JSON.parse((await env.SONGS.get("debugLog")) || "[]"); const entry = { time: new Date().toISOString(), incomingChatId, configuredChannelId: env.CHANNEL_ID || null, matched, hasAudio }; debugLog.unshift(entry); debugLog = debugLog.slice(0, 8); await env.SONGS.put("debugLog", JSON.stringify(debugLog)); // فقط پیام‌های خودِ کانال پذیرفته می‌شوند if (!matched) { return new Response("ignored"); } if (!hasAudio) { return new Response("no audio, ignored"); } const audio = msg.audio; if (audio.file_size && audio.file_size > MAX_BYTES) { entry.skippedTooLarge = true; debugLog[0] = entry; await env.SONGS.put("debugLog", JSON.stringify(debugLog)); return new Response("skipped: file too large for bot API (>20MB)"); } const uniqueId = audio.file_unique_id; // اگر این فایل قبلاً آپلود شده، دوباره تکرار نمی‌شود const already = await env.SONGS.get(`uploaded:${uniqueId}`); if (already) { return new Response("already uploaded, skipped"); } try { // ۱) گرفتن مسیر فایل از تلگرام const tgRes = await fetch(`https://api.telegram.org/bot${env.BOT_TOKEN}/getFile?file_id=${audio.file_id}`); const tgData = await tgRes.json(); if (!tgData.ok) throw new Error("telegram getFile failed: " + JSON.stringify(tgData)); // ۲) دانلود خودِ فایل از تلگرام const fileUrl = `https://api.telegram.org/file/bot${env.BOT_TOKEN}/${tgData.result.file_path}`; const fileRes = await fetch(fileUrl); if (!fileRes.ok) throw new Error("telegram file download failed"); const buffer = await fileRes.arrayBuffer(); const base64Content = arrayBufferToBase64(buffer); // ۳) ساخت اسم فایل و آپلود به گیت‌هاب const title = audio.title || (audio.file_name ? audio.file_name.replace(/\.[^/.]+$/, "") : "بدون عنوان"); const artist = audio.performer || ""; const fileName = buildFileName(artist, title, uniqueId, msg.date || Math.floor(Date.now() / 1000)); const githubPath = `${SONGS_PATH}/${encodeURIComponent(fileName)}`; const githubApi = `https://api.github.com/repos/${GITHUB_OWNER}/${GITHUB_REPO}/contents/${githubPath}`; const putRes = await fetch(githubApi, { method: "PUT", headers: { "Authorization": `token ${env.GITHUB_TOKEN}`, "Accept": "application/vnd.github+json", "User-Agent": "car-music-bot" }, body: JSON.stringify({ message: `افزودن آهنگ: ${fileName}`, content: base64Content, branch: GITHUB_BRANCH }) }); if (!putRes.ok) { const errText = await putRes.text(); throw new Error("github upload failed: " + errText); } // ۴) ثبت این‌که این فایل دیگر آپلود شده (جلوگیری از تکرار در فورواردهای بعدی) await env.SONGS.put(`uploaded:${uniqueId}`, fileName); return new Response("uploaded to github: " + fileName); } catch (e) { entry.error = String(e); debugLog[0] = entry; await env.SONGS.put("debugLog", JSON.stringify(debugLog)); return new Response("error: " + String(e), { status: 500 }); } } return new Response("radio backend (github relay) is running ✅", { status: 200 }); } };
